{\bfseries Algorithm 3} Inverse Kinematics obtained by optimization $\textcolor{red}{\bm{u}_a} = f^{-1}(\textcolor{green!50!black}{\bm{y}_e})$
\vspace{0.3em}
\hrule
\vspace{0.5em}

\begin{algorithmic}[1]

\State{Initialize $\bm{q}^{0} = [\bm{q}_1^{0};\, \bm{q}_2^{0};\, \bm{q}_3^{0};\, \bm{q}_4^{0}]$}
\For{$i=1, \ldots, \mathrm{maxiter}$}
\State{Form the residual:}\Comment{simulate states and gradients}
    \[
    \left\{
    \begin{aligned}
   \bm{b}_1 \leftarrow \bm{F}_1(\bm{q}_1^{i-1}) &+ \bm{M}_1 \bm{g} \\
    &\vdots \\
   \bm{b}_4 \leftarrow \bm{F}_4(\bm{q}_4^{i-1}) &+ \bm{M}_4 \bm{g}
    \end{aligned}
    \right.
    %\tag{14}
    \]

    \State{Compute $\bm{A}_j(\bm{q}_j^{i-1})$, $\bm{H}_j$ ($j \in [1,4]$), $\bm{A}$, and $\bm{H}_a, \bm{H}_e, \bm{\delta}_a(\bm{q}^{i-1})$}
    \State{Find $d\bm{q}$:}
    \Comment{QP solve}
    \[
    \left\{
    \begin{aligned}
      \min_{\bm{\lambda}_a,d\bm{q}} \, & \frac{1}{2}\| \bm{\delta}_e(\bm{q}_{i-1}) + \bm{H}_e d\bm{q} - \bm{y}_e\|^2 \\
      \text{s.t. } & \bm{A}d\bm{q} &= \bm{b} + \bm{H}_e^\top\bm{\lambda}_e + \bm{H}_a^\top\bm{\lambda}_a \\
    \end{aligned}
    \right.
    \tag{20}
    \]

    \State{$\bm{q}^{i} \leftarrow \bm{q}^{i-1} + \alpha d\bm{q}$\Comment{take solver step}}
    \If{$\lVert \bm{q}^{i} - \bm{q}^{i-1} \rVert < \epsilon$}
    \State{\Return: $\textcolor{red}{\bm{u}_a} = \bm{\delta}_a(\bm{q}^{i-1}) + \bm{H}_a d\bm{q}$\Comment{solver converged}}
    \EndIf{}
\EndFor%

\end{algorithmic}
